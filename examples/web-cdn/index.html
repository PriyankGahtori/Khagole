Direct CDN Integration

Load the engine directly from jsDelivr in any browser application: 

<script type="module">
  import init, { Engine, calcPanchang, calcDivisional, calcVimshottariDasha, nakshatraInfo } 
    from 'https://cdn.jsdelivr.net/npm/khagole-engine@0.0.12/khagole_engine_wasm.js';

  // 1. Initialize WebAssembly runtime directly from jsDelivr CDN
  await init();

  // 2. Fetch NASA JPL ephemeris kernel
  const res = await fetch('https://priyankgahtori.github.io/Khagole/ephemeris/de440s-1920-2070.bsp');
  const kernelBytes = new Uint8Array(await res.arrayBuffer());

  const engine = new Engine();
  engine.addKernel('de440s-1920-2070.bsp', kernelBytes);

  // 3. Compute chart with canonical Parashari defaults
  const chart = engine.chart({
    utc: '1955-02-25T03:15:00Z',
    latitude: 37.7749,
    longitude: -122.4194,
    zodiac_type: 'sidereal',
    ayanamsa: 'lahiri',
    position_mode: 'true',
    node_type: 'mean',
    house_system: 'W',
  });
</script>
